TABLE 
    file.folder as "Folder",
    file.ctime as "Created",
    file.mtime as "Modified",
    length(file.inlinks) as "Backlinks"
FROM ""
WHERE 
    (contains(file.name, "Home Assistant") 
    OR contains(file.name, "home-assistant")
    OR contains(file.name, "HomeAssistant")
    OR contains(file.tags, "#home-assistant")
    OR contains(file.tags, "#homeassistant")
    OR contains(string(file.outlinks), "Home Assistant")
    OR contains(string(this.file.content), "Home Assistant"))
    AND !contains(string(file.outlinks), "Home-Assistant MoC")
    AND file.name != "Home-Assistant MoC"
SORT file.mtime DESC
