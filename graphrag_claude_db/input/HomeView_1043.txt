### Notes
-  Minimal Homekit view.  It displays a list of homes
```
struct ContentView: View {
    @StateObject var homeStore: HomeStore

    init(homeManager: HMHomeManager) {
        _homeStore = StateObject(wrappedValue: HomeStore(homeManager: homeManager))
    }

    var body: some View {
        List(homeStore.homes, id: \.uniqueIdentifier) { home in
            Text(home.name)
        }
    }
}
```
