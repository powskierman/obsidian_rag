### Step 3: Create Embedding Service (5 minutes)

This service stores and searches your documents.

**Create `embedding_service.py`:**
```python
from flask import Flask, request, jsonify
import chromadb
from sentence_transformers import SentenceTransformer

app = Flask(__name__)

# Initialize Chroma and embedding model
client = chromadb.Client()
collection = client.get_or_create_collection("knowledge_base")
model = SentenceTransformer('all-MiniLM-L6-v2')

@app.route('/add', methods=['POST'])
def add_document():
    """Add a document to the knowledge base"""
    data = request.json
    
    # Generate embedding
    embedding = model.encode(data['text']).tolist()
    
    # Store in Chroma
    collection.add(
        documents=[data['text']],
        ids=[data['id']],
        embeddings=[embedding],
        metadatas=[data.get('metadata', {})]
    )
    
    return jsonify({
        "status": "success",
        "id": data['id']
    })

@app.route('/query', methods=['POST'])
def query():
    """Search the knowledge base"""
    data = request.json
    
    # Generate query embedding
    embedding = model.encode(data['query']).tolist()
    
    # Search Chroma
    results = collection.query(
        query_embeddings=[embedding],
        n_results=data.get('n_results', 3)
    )
    
    return jsonify(results)

@app.route('/health', methods=['GET'])
def health():
    """Health check endpoint"""
    return jsonify({"status": "healthy"})

if __name__ == '__main__':
    print("🚀 Embedding service starting on http://localhost:8000")
    app.run(host='0.0.0.0', port=8000, debug=False)
```

### Step 4: Create Gradio Frontend (5 minutes)

**Create `app.py`:**
```python
import gradio as gr
import requests
import json

def add_sample_data():
    """Add sample documents to knowledge base"""
    sample_docs = [
        {
            "id": "doc1",
            "text": "RAG (Retrieval-Augmented Generation) combines retrieval systems with language models to provide accurate, context-aware responses.",
            "metadata": {"source": "tutorial"}
        },
        {
            "id": "doc2",
            "text": "Vector databases store embeddings (numerical representations) of text, enabling semantic search that understands meaning, not just keywords.",
            "metadata": {"source": "tutorial"}
        },
        {
            "id": "doc3",
            "text": "Ollama is an open-source tool that lets you run large language models locally on your computer without API costs.",
            "metadata": {"source": "tutorial"}
        }
    ]
    
    for doc in sample_docs:
        requests.post('http://localhost:8000/add', json=doc)
    
    return "✅ Sample data loaded!"

def chat(message, history):
    """Handle chat messages"""
    try:
        # Step 1: Get relevant context from knowledge base
        context_response = requests.post(
            'http://localhost:8000/query',
            json={'query': message, 'n_results': 3}
        )
        
        contexts = context_response.json().get('documents', [[]])[0]
        context_text = "\\n\\n".join(contexts) if contexts else ""
        
        # Step 2: Query Ollama with context
        system_prompt = "You are a helpful assistant. Use the following context to answer the question. If the context doesn't contain relevant information, say so."
        
        if context_text:
            system_prompt += f"\\n\\nContext:\\n{context_text}"
        
        ollama_response = requests.post(
            'http://localhost:11434/api/chat',
            json={
                'model': 'llama3.2',
                'messages': [
                    {'role': 'system', 'content': system_prompt},
                    {'role': 'user', 'content': message}
                ],
                'stream': False
            }
        )
        
        return ollama_response.json()['message']['content']
        
    except Exception as e:
        return f"❌ Error: {str(e)}\\n\\nMake sure embedding service is running!"

# Create interface
with gr.Blocks(title="RAG Knowledge Base", theme="soft") as demo:
    gr.Markdown("# 🧠 RAG Knowledge Base")
    gr.Markdown("Ask questions about the knowledge base!")
    
    with gr.Row():
        with gr.Column():
            load_btn = gr.Button("📚 Load Sample Data", variant="primary")
            status_msg = gr.Textbox(label="Status", interactive=False)
    
    chatbot = gr.ChatInterface(
        fn=chat,
        examples=[
            "What is RAG?",
            "How do vector databases work?",
            "Tell me about Ollama"
        ],
        title=None
    )
    
    load_btn.click(fn=add_sample_data, outputs=status_msg)

if __name__ == "__main__":
    print("🚀 Starting RAG Knowledge Base...")
    print("📝 Open http://localhost:7860 in your browser")
    demo.launch(
        server_name="0.0.0.0",
        server_port=7860,
        share=False
    )
```

## What's Happening Under the Hood?

### The RAG Pipeline

```
┌─────────────────┐
│ 1. User Query   │  "What is RAG?"
└────────┬────────┘
         │
         ↓
┌─────────────────┐
│ 2. Embedding    │  Convert to vector [0.1, 0.2, ...]
└────────┬────────┘
         │
         ↓
┌─────────────────┐
│ 3. Search Chroma│  Find similar documents
└────────┬────────┘
         │
         ↓
┌─────────────────┐
│ 4. Get Context  │  "RAG combines retrieval..."
└────────┬────────┘
         │
         ↓
┌─────────────────┐
│ 5. Query Ollama │  LLM + Context → Answer
└────────┬────────┘
         │
         ↓
┌─────────────────┐
│ 6. Response     │  Formatted answer
└─────────────────┘
```

### How Embeddings Work

```python
# Text → Vector (simplified)
"What is RAG?" → [0.123, -0.456, 0.789, ...]  # 384 dimensions

# Similar texts have similar vectors
"What is RAG?" → [0.123, -0.456, 0.789, ...]
"Explain RAG"  → [0.119, -0.449, 0.791, ...]  # Very similar!
"Weather today"→ [0.876, 0.234, -0.123, ...]  # Different
```

## Customization Guide

### Add Your Own Documents

**Method 1: API (Programmatic)**
```python
import requests

def add_document(text, doc_id, metadata=None):
    response = requests.post(
        'http://localhost:8000/add',
        json={
            'id': doc_id,
            'text': text,
            'metadata': metadata or {}
        }
    )
    return response.json()

# Example: Add product documentation
add_document(
    text="Our product supports OAuth 2.0 authentication...",
    doc_id="doc_auth",
    metadata={"category": "security", "version": "2.0"}
)
```

**Method 2: Bulk Upload**
```python
# bulk_upload.py
import requests

def upload_file(filepath):
    with open(filepath, 'r') as f:
        content = f.read()
    
    # Split into chunks (optional, for long documents)
    chunks = [content[i:i+1000] for i in range(0, len(content), 1000)]
    
    for idx, chunk in enumerate(chunks):
        requests.post(
            'http://localhost:8000/add',
            json={
                'id': f"{filepath}_chunk_{idx}",
                'text': chunk,
                'metadata': {'source': filepath, 'chunk': idx}
            }
        )

# Usage
upload_file('docs/manual.txt')
```

### Adjust Response Quality

```python
# In app.py, modify Ollama call:
ollama_response = requests.post(
    'http://localhost:11434/api/chat',
    json={
        'model': 'llama3.2',
        'messages': [...],
        'stream': False,
        'options': {
            'temperature': 0.7,    # Lower = more focused (0.1-1.0)
            'top_p': 0.9,         # Nucleus sampling
            'top_k': 40,          # Top-k sampling
            'num_ctx': 4096       # Context window size
        }
    }
)
```

## Troubleshooting

### Issue: "Connection refused" on port 8000
**Solution:**
```bash
# Check if embedding service is running
curl http://localhost:8000/health

# If not, start it:
python embedding_service.py
```

### Issue: "Ollama not found"
**Solution:**
```bash
# Verify Ollama is running
ollama list

# If not installed:
curl -fsSL https://ollama.com/install.sh | sh
```

### Issue: Slow responses
**Solutions:**
1. Use smaller model: `ollama pull llama3.2:3b`
2. Reduce context window in Ollama call
3. Limit search results: `'n_results': 2`

### Issue: Out of memory
**Solutions:**
1. Use smaller model: `phi3:3.8b` (2.3GB vs 4GB)
2. Close other applications
3. Reduce context window

### Issue: Wrong answers
**Solutions:**
1. Add more relevant documents
2. Improve document chunking
3. Increase search results: `'n_results': 5`
4. Adjust temperature: lower = more factual
