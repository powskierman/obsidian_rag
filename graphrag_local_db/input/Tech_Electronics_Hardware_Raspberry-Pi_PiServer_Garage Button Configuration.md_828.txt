### Notes
- Use [Custom Button Card](https://github.com/custom-cards/button-card) Installed with HACS
- This is the script:
```yaml TI:"Button Card yaml" FOLD
type: custom:button-card
entity: switch.kona_left_garage_door_switch
icon: mdi:garage
name: Garage
show_state: false
state:
  - operator: template
    value: >
      [[[ return states['switch.kona_left_garage_door_switch'].state === 'on'
      ]]]
    icon: mdi:garage
    color: orange
  - operator: template
    value: >
      [[[ return states['binary_sensor.kona_left_door_sensor'].state === 'on'
      ]]]
    icon: mdi:garage-open
    color: red
    label: Open
  - operator: template
    value: >
      [[[ return states['binary_sensor.kona_left_door_sensor'].state === 'off'
      ]]]
    icon: mdi:garage
    color: green
    label: Closed
tap_action:
  action: call-service
  service: script.toggle_left_door
```

```yaml TI:"toggle_left_door Script" FOLD
alias: Toggle Left Door
sequence:
  - service: switch.toggle
    target:
      entity_id: switch.kona_left_garage_door_switch
  - delay:
      milliseconds: 500
  - service: switch.toggle
    target:
      entity_id: switch.kona_left_garage_door_switch
mode: single
icon: mdi:toggle-switch-off-outline
```
