### Notes
t_start = timer start time

T_Start = Starting temperature

Duration = millis() - t_start

preheat_rate = temperature increase per second

T_preheat = final destination temperature

t_final = projected final time based on temperature points and temperature rates



t_final = (T_preheat - T_start) / preheat_rate + t_start;
