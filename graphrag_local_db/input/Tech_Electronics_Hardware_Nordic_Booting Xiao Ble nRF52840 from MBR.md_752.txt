### Notes
- Create a xiao_ble.overlay file that modifies the `code_partition` node in the `flash0`'s partitions.:
```
&flash0 {
	partitions {
		compatible = "fixed-partitions";
		#address-cells = <1>;
		#size-cells = <1>;

		code_partition: partition@27000 {
			reg = <0x00001000 0x000FF000>;
		};
	};
};
```
- Load the overlay file from CmakeLists.txt:
```
cmake_minimum_required(VERSION 3.20.0)
set(DTC_OVERLAY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/xiao_ble.overlay")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(blinky)
target_sources(app PRIVATE src/main.c)	
```
