<%*
let title = '';
// Try to get highlighted text first
const highlightedText = await tp.file.selection();
if (highlightedText) {
    title = highlightedText;
} else {
    // If no highlight, try clipboard
    title = await tp.system.clipboard();
}

// If both highlight and clipboard are empty, prompt for manual input
if (!title) {
    title = await tp.system.prompt("Enter a title for the note:");
}

// Clean up the title
title = title.split('\n')[0].trim().substring(0, 50);

// Rename the file
if (title) {
    await tp.file.rename(title);
} else {
    // If still no title, use a default
    title = "Untitled Note";
    await tp.file.rename(title);
}
-%>
