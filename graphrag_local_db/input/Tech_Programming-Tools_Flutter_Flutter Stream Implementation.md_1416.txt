```
import 'dart:async';

StreamController<Color> _controller = StreamController<Color>();
Stream get onAlarmColor => _controller.stream;

void sendAlarmColor() {
  onAlarmColor.listen(
    (newColor) {
      SwitchClass.alarmColor;
      Icon(Icons.alarm, size: 175, color: SwitchClass.alarmColor);
      print(SwitchClass.alarmColor);
    },
  );
} 

_controller.add(alarmColor);
```
